### 0.3 Copy rules for Round 3
- Never say an unsupported claim is "false". Semantic matches say "similar to rule X — check it".
- The percentage on a semantic badge is **similarity to the rule's wording**, not the probability of a violation (tooltip says so).
- Contrast/palette numbers are measurements; say "measured", show the threshold.
- Don't claim WCAG conformance: "contrast checked against WCAG AA 4.5:1".
- Colour is never the only signal: severity icon + text; bbox stroke style varies by check type (solid = palette, dashed = contrast).
- Notice near upload/ingest: "AI features use a free-tier external model. Use non-confidential sample material only."

export async function apiUpload<T>(path: string, form: FormData): Promise<T> {
  const res = await fetch(`${API_URL}${path}`, { method: 'POST', headers: authHeaders(false), body: form });
  return handle<T>(res);              // same {"error":{code,message,details}} → ApiError mapping as apiFetch
}
export async function apiBlob(path: string): Promise<Blob> {
  const res = await fetch(`${API_URL}${path}`, { headers: authHeaders(false) });
  if (!res.ok) throw await toApiError(res);
  return res.blob();
}
export async function apiDelete(path: string): Promise<void> { /* DELETE, expects 204 */ }

### 4.6 Hooks
```ts
// useDocumentPolling: poll while status is queued|parsing|extracting
export const useDocument = (id?: number) => useQuery({
  queryKey: ['document', id], queryFn: () => getDocument(id!), enabled: !!id,
  refetchInterval: q => ['queued','parsing','extracting'].includes(q.state.data?.status ?? '') ? 2000 : false,
});
// useAssetBlobUrl: fetch with bearer → object URL; revoke on unmount
export function useAssetBlobUrl(assetId?: number) {
  const [url, setUrl] = useState<string>();
  useEffect(() => { if (!assetId) return; let u: string; let dead = false;
    assetContent(assetId).then(b => { if (!dead) { u = URL.createObjectURL(b); setUrl(u); } });
    return () => { dead = true; if (u) URL.revokeObjectURL(u); }; }, [assetId]);
  return url;
}

    <figure className="relative w-full overflow-hidden rounded-xl border border-white/10 bg-black/40"
            style={{ aspectRatio: `${asset.width} / ${asset.height}`, maxHeight: '70vh' }}>
      {src ? <img src={src} alt={`Creative: ${asset.filename}`} className="absolute inset-0 h-full w-full select-none" draggable={false} />
           : <div className="absolute inset-0 animate-pulse bg-white/5" aria-busy="true" />}
      {analyzing && !reduce && <div className="scan-line" aria-hidden />}
      <svg className="absolute inset-0 h-full w-full" viewBox="0 0 1000 1000" preserveAspectRatio="none"
           role="group" aria-label="Detected issues on the image">
        {findings.filter(f => layers[f.check_type]).map(f => (
          <BboxOverlay key={f.id} finding={f} active={f.id === activeFindingId}
                       dimmed={activeFindingId != null && f.id !== activeFindingId} onSelect={onSelect} />
        ))}
      </svg>
    </figure>
  );
}
```
> `object-fit` is not needed: the figure's `aspect-ratio` equals the image's, so `h-full w-full` fills it exactly, and `preserveAspectRatio="none"` maps the 0–1000 box onto it.
